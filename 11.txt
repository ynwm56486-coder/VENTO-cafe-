                            <img src="${item.image || 'https://images.unsplash.com/photo-1546069901-ba9599a7e63c?auto=format&fit=crop&w=600&q=80'}" alt="${item.title}" class="w-full h-full object-cover group-hover:scale-105 transition duration-500">
                            ${item.badge ? `<span class="absolute top-3 left-3 bg-black/70 backdrop-blur-md text-vento-gold border border-vento-gold/30 text-[10px] font-bold px-2.5 py-1 rounded-full uppercase">${item.badge}</span>` : ''}
                            
                            ${isAdminLoggedIn ? `
                                <div class="absolute top-3 right-3 flex gap-1">
                                    <button onclick="openEditModal('${item.id}')" class="w-8 h-8 rounded-full bg-vento-gold text-vento-dark flex items-center justify-center text-xs font-bold shadow-lg hover:bg-yellow-400">
                                        <i class="fa-solid fa-pencil"></i>
                                    </button>
                                    <button onclick="deleteMenuItem('${item.id}')" class="w-8 h-8 rounded-full bg-vento-red text-white flex items-center justify-center text-xs font-bold shadow-lg hover:bg-red-600">
                                        <i class="fa-solid fa-trash"></i>
                                    </button>
                                </div>
                            ` : ''}
                        </div>
                        <div class="p-5">
                            <div class="flex justify-between items-start mb-2">
                                <h4 class="font-serif font-bold text-lg text-white group-hover:text-vento-gold transition">${item.title}</h4>
                                <span class="font-serif font-bold text-vento-gold text-lg">$${parseFloat(item.price).toFixed(2)}</span>
                            </div>
                            <p class="text-xs text-gray-400 leading-relaxed">${item.desc}</p>
                        </div>
                    </div>
                `;
                grid.appendChild(card);
            });
        }

        function updateWifiDisplay() {
            document.getElementById('wifi-ssid-display').textContent = wifiSettings.ssid;
            document.getElementById('wifi-pass-display').textContent = wifiSettings.pass;

            const qrContainer = document.getElementById('qrcode');
            qrContainer.innerHTML = '';
            
            const wifiString = `WIFI:S:${wifiSettings.ssid};T:WPA;P:${wifiSettings.pass};;`;
            
            new QRCode(qrContainer, {
                text: wifiString,
                width: 70,
                height: 70,
                colorDark : "#000000",
                colorLight : "#ffffff",
                correctLevel : QRCode.CorrectLevel.H
            });
        }

        function copyWifiPassword() {
            navigator.clipboard.writeText(wifiSettings.pass);
            alert('Wi-Fi Password copied to clipboard: ' + wifiSettings.pass);
        }

            e.preventDefault();
            const input = document.getElementById('admin-pass-input').value;

            }

        function addNewDish() {
            const title = document.getElementById('new-item-title').value;
            const price = parseFloat(document.getElementById('new-item-price').value);
            const category = document.getElementById('new-item-category').value;
            const image = document.getElementById('new-item-image').value;
            const desc = document.getElementById('new-item-desc').value;

            if (!title || isNaN(price)) {
                alert('Please enter a valid title and price.');
                return;
            }

            const newItem = {
                id: Date.now().toString(),
                title,
                price,
                category,
                image: image || 'https://images.unsplash.com/photo-1546069901-ba9599a7e63c?auto=format&fit=crop&w=600&q=80',
                desc: desc || 'Delicious dish freshly prepared at Vento.',
                badge: 'New'
            };

            menuItems.unshift(newItem);
            saveMenuToStorage();
            renderMenu();

            document.getElementById('new-item-title').value = '';
            document.getElementById('new-item-price').value = '';
            document.getElementById('new-item-image').value = '';
            document.getElementById('new-item-desc').value = '';

            alert('Dish added successfully!');
        }

        function deleteMenuItem(id) {
            if (confirm('Are you sure you want to delete this dish from the menu?')) {
                menuItems = menuItems.filter(item => item.id !== id);
                saveMenuToStorage();
                renderMenu();
            }
        }

        function openEditModal(id) {
            const item = menuItems.find(i => i.id === id);
            if (!item) return;

            document.getElementById('edit-dish-id').value = item.id;
            document.getElementById('edit-dish-title').value = item.title;
            document.getElementById('edit-dish-price').value = item.price;
            document.getElementById('edit-dish-desc').value = item.desc;

            document.getElementById('edit-dish-modal').classList.remove('hidden');
        }

        function closeEditModal() {
            document.getElementById('edit-dish-modal').classList.add('hidden');
        }

        function saveDishEdit() {
            const id = document.getElementById('edit-dish-id').value;
            const item = menuItems.find(i => i.id === id);

            if (item) {
                item.title = document.getElementById('edit-dish-title').value;
                item.price = parseFloat(document.getElementById('edit-dish-price').value) || item.price;
                item.desc = document.getElementById('edit-dish-desc').value;

                saveMenuToStorage();
                renderMenu();
                closeEditModal();
            }
        }

        function saveWifiSettings() {
            const ssid = document.getElementById('edit-wifi-ssid').value;
            const pass = document.getElementById('edit-wifi-pass').value;

            if (ssid && pass) {
                wifiSettings = { ssid, pass };
                localStorage.setItem('vento_wifi', JSON.stringify(wifiSettings));
                updateWifiDisplay();
                alert('Wi-Fi credentials updated successfully!');
            }
        }

        function changeAdminPassword() {
            const newPass = document.getElementById('new-admin-pass').value;
            if (newPass) {
                adminPassword = newPass;
                localStorage.setItem('vento_admin_pass', newPass);
                alert('Admin Password updated successfully!');
                document.getElementById('new-admin-pass').value = '';
            }
        }
